bool isOLEDConnected() {
  Wire.beginTransmission(OLED_ADDR);
  return (Wire.endTransmission() == 0); // trả về true nếu thiết bị phản hồi
}

void sendCommand(uint8_t cmd) {
  Wire.beginTransmission(OLED_ADDR);
  Wire.write(0x00);
  Wire.write(cmd);
  Wire.endTransmission();
}

void sendData(uint8_t data) {
  Wire.beginTransmission(OLED_ADDR);
  Wire.write(0x40);
  Wire.write(data);
  Wire.endTransmission();
}
// Khoi tao OLED
void initOLED() {
  delay(100);
  sendCommand(0xAE); // Tat man hinh
  sendCommand(0xD5); sendCommand(0x80);
  sendCommand(0xA8); sendCommand(0x3F);
  sendCommand(0xD3); sendCommand(0x00);
  sendCommand(0x40);
  sendCommand(0x8D); sendCommand(0x14);
  sendCommand(0x20); sendCommand(0x00);
  sendCommand(0xA0);  // Segment Remap - mặc định
  sendCommand(0xC0);  // COM Output Scan Direction - mặc định
  sendCommand(0xDA); sendCommand(0x12);
  sendCommand(0x81); sendCommand(0xCF);
  sendCommand(0xD9); sendCommand(0xF1);
  sendCommand(0xDB); sendCommand(0x40);
  sendCommand(0xA4);
  sendCommand(0xA6);
  sendCommand(0xAF); // Bat man hinh
}
// Xoa man hinh OLED
void clearScreen() {
  if (!oledAvailable) return;
  for (uint8_t page = 0; page < 8; page++) {
    sendCommand(0xB0 + page);
    sendCommand(0x00);
    sendCommand(0x10);
    for (uint8_t col = 0; col < 128; col++) {
      sendData(0x00);
    }
  }
}

void drawChar(char c) {
  if (c >= 'a' && c <= 'z') c -= 32; // chuyển sang viết hoa

  uint8_t index;

  if (c == ' ')      index = 0;
  else if (c >= '0' && c <= '9') index = c - '0' + 1;
  else if (c == ':') index = 11;
  else if (c >= 'A' && c <= 'Z') index = c - 'A' + 12;
  else return; // không hỗ trợ ký tự này

  for (int i = 0; i < 5; i++) sendData(font5x7[index][i]);
  sendData(0x00); // khoảng cách giữa các ký tự
}

void drawText(uint8_t col, uint8_t page, const char* text) {
  if (!oledAvailable) return;  // Bỏ qua nếu không có OLED
  sendCommand(0xB0 + page);
  sendCommand(0x00 + (col & 0x0F));
  sendCommand(0x10 + ((col >> 4) & 0x0F));
  while (*text) {
    drawChar(*text++);
  }
}

void updateOLEDStatus(const char* wifi, const char* data, const char* post, const char* led) {
  if (!oledAvailable) return;
  clearScreen();
  drawText(4, 0, wifi);
  drawText(4, 1, data);
  drawText(4, 2, post);
  drawText(4, 3, led);
}
void buildDataSend(char* buffer, size_t bufferSize,
                   const char* serial,
                   int ua, float ia, int pa,
                   int ub, float ib, int pb,
                   int uc, float ic,
                   int energy) {
  snprintf(buffer, bufferSize,
           "SN%sUA%03dIA%04.1fPA%03dUB%03dIB%04.1fPB%03dUC%03dIC%04.1fE%05dP8\n",
           serial,
           ua, ia, pa,
           ub, ib, pb,
           uc, ic,
           energy);
}

void setup() {
  Serial.begin(115200);
  mySerial.begin(BAUD_RATE, SERIAL_8N1, RX_PIN, TX_PIN);

  pinMode(GREEN_LED_PIN, OUTPUT);
  pinMode(RED_LED_PIN, OUTPUT);
  digitalWrite(GREEN_LED_PIN, LOW);
  digitalWrite(RED_LED_PIN, HIGH);
  isGreenOn = false;
  isRedOn = true;

  // Bắt đầu I2C và giảm tốc độ xuống 100kHz để tránh lỗi font chữ
  Wire.begin(SDA_PIN, SCL_PIN);
  Wire.setClock(100000);  // Ổn định giao tiếp I2C

  // Kiểm tra OLED có kết nối không
  oledAvailable = isOLEDConnected();

  if (oledAvailable) {
    initOLED();
    clearScreen();
    drawText(0, 0, "Connecting...");
    Serial.println("OLED connected.");
  } else {
    Serial.println("OLED not found, continuing without display.");
  }

  // Kết nối WiFi
  WiFi.begin(ssid, password);
  Serial.print("Connecting to WiFi");

  int attempts = 0;
  while (WiFi.status() != WL_CONNECTED && attempts < 10) {
    delay(500);
    Serial.print(".");
    attempts++;
  }

  if (WiFi.status() == WL_CONNECTED) {
    Serial.println("\nWiFi connected! IP: " + WiFi.localIP().toString());
    updateOLEDStatus("WIFI OK", "NO DATA", "---", "LED: RED");
  } else {
    Serial.println("\nWiFi connection failed.");
    updateOLEDStatus("WIFI FAIL", "NO DATA", "---", "LED: RED");
  }
}

void loop() {
  readUARTData();

  // === 1. Kiểm tra lại OLED mỗi 5 giây để hỗ trợ hot-swap ===
  static unsigned long lastOLEDCheck = 0;
  if (millis() - lastOLEDCheck > 5000) {
    lastOLEDCheck = millis();
    if (!oledAvailable && isOLEDConnected()) {
      Serial.println("OLED reconnected → initializing...");
      oledAvailable = true;
      initOLED();
      clearScreen();
      updateOLEDStatus("WIFI OK", "NO DATA", "---", isGreenOn ? "LED: GREEN" : "LED: RED");
    }
  }
   
  // === 2. Kiểm tra trạng thái WiFi mỗi 10 giây ===
  static unsigned long lastWiFiCheck = 0;
  static bool wasWiFiConnected = (WiFi.status() == WL_CONNECTED);

  if (millis() - lastWiFiCheck > 10000) {
    lastWiFiCheck = millis();
    bool nowConnected = (WiFi.status() == WL_CONNECTED);

    if (wasWiFiConnected && !nowConnected) {
      Serial.println("WiFi disconnected!");
      updateOLEDStatus("WIFI FAIL", "NO DATA", "---", isGreenOn ? "LED: GREEN" : "LED: RED");
    }

    if (!wasWiFiConnected && nowConnected) {
      Serial.println("WiFi reconnected!");
      updateOLEDStatus("WIFI OK", "NO DATA", "---", isGreenOn ? "LED: GREEN" : "LED: RED");
    }

    wasWiFiConnected = nowConnected;
  }

  // === 3. Nếu có dữ liệu mới từ UART ===
  if (dataReady) {
    Serial.println("New data received: " + currentData);
    bool postSuccess = sendHTTPPost(currentData);

    const char* wifiStatusText = (WiFi.status() == WL_CONNECTED) ? "WIFI OK" : "WIFI FAIL";

    if (postSuccess) {
      Serial.println("Data sent successfully!");
      updateOLEDStatus(wifiStatusText, "DATA OK", "POST OK", isGreenOn ? "LED: GREEN" : "LED: RED");
    } else {
      Serial.println("Failed to send data!");
      updateOLEDStatus(wifiStatusText, "DATA OK", "POST FAIL", isGreenOn ? "LED: GREEN" : "LED: RED");
    }

    previousData = currentData;
    dataReady = false;
  }
for (int i = 0; i < 3; i++) {
    char dataSend[100];
    buildDataSend(dataSend, sizeof(dataSend),
                  deviceSNs[i],
                  220.0 + i, 5.0 + i * 0.1, 120.0 + i * 2,
                  221.0 + i, 5.1 + i * 0.1, 122.0 + i * 2,
                  219.0 + i, 5.2 + i * 0.1,
                  4567 + i);

    Serial.print("Data to send: ");
    Serial.println(dataSend);

    sendHTTPPost(dataSend);  // gửi từng thiết bị một
}

  // === 4. Kiểm tra timeout Zigbee (không có tín hiệu mới) ===
  if (millis() - lastSignalTime > signalTimeout) {
    if (!isRedOn) {
      digitalWrite(GREEN_LED_PIN, LOW);
      digitalWrite(RED_LED_PIN, HIGH);
      Serial.println("No Zigbee signal → RED ON");

      isGreenOn = false;
      isRedOn = true;

      const char* wifiStatusText = (WiFi.status() == WL_CONNECTED) ? "WIFI OK" : "WIFI FAIL";
      updateOLEDStatus(wifiStatusText, "NO DATA", "---", "LED: RED");
    }
  }

  delay(1000); // Đợi 1 giây trước khi lặp lại
}

void readUARTData() {
  static String buffer = "";

  while (mySerial.available()) {
    char c = mySerial.read();

    if (c == '\n') {
      currentData = buffer;
      currentData.trim();

      if (currentData != previousData && currentData.length() > 0) {
        dataReady = true;

        if (!isGreenOn) {
          digitalWrite(GREEN_LED_PIN, HIGH);
          digitalWrite(RED_LED_PIN, LOW);
          Serial.println("GREEN ON, RED OFF");

          isGreenOn = true;
          isRedOn = false;
        }

        lastSignalTime = millis();
      }

      buffer = "";
    } else if (c != '\r') {
      buffer += c;

      if (buffer.length() > 100) {
        buffer = "";
        Serial.println("Buffer overflow");
      }
    }
  }
}

bool sendHTTPPost(String data) {
  data.trim();  // Xóa ký tự thừa như '\n' hoặc khoảng trắng

  if (data.length() < 10) {
    Serial.println("Data too short");
    return false;
  }

  if (WiFi.status() != WL_CONNECTED) {
    Serial.println("WiFi not connected");
    return false;
  }

  HTTPClient http;
  http.setTimeout(5000); // timeout 5 giây

  http.begin(apiURL);
  http.setFollowRedirects(HTTPC_FORCE_FOLLOW_REDIRECTS);

  http.addHeader("Content-Type", "application/json");
  String jsonPayload = "{\"data\":\"" + data + "\"}";

  int httpResponseCode = http.POST(jsonPayload);
  String response = http.getString();

  Serial.println("HTTP Response: " + String(httpResponseCode));
  Serial.println("Response: " + response);

  // Nếu server không hỗ trợ POST, thử lại với GET
  if (httpResponseCode == 405) {
    http.end();
    Serial.println("POST bị từ chối (405), thử lại với GET...");

    String urlWithQuery = String(apiURL) + "?data=" + data;
    http.begin(urlWithQuery);
    httpResponseCode = http.GET();
    response = http.getString();

    Serial.println("HTTP Response (GET): " + String(httpResponseCode));
    Serial.println("Response: " + response);
  }

  http.end();
  return httpResponseCode == 200 || httpResponseCode == 201;
}
